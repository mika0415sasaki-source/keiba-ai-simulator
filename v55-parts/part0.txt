function oddsNumber(x){
 if(x==null)return null;
 if(Number.isFinite(+x)&&+x>0)return +x;
 if(typeof x!=="object")return null;
 const keys=["odds","win_odds","odds_min","min","value","price"];
 for(const k of keys){if(Number.isFinite(+x[k])&&+x[k]>0)return +x[k]}

 const lo=+[x.odds_min,x.min,x.low].find(v=>Number.isFinite(+v)&&+v>0);
 const hi=+[x.odds_max,x.max,x.high].find(v=>Number.isFinite(+v)&&+v>0);
 if(Number.isFinite(lo)&&Number.isFinite(hi))return (lo+hi)/2;
 return null;
}

 // Supabaseのrace_id単位集計を最優先。Safariや別URLへ移っても累積を失わない。

function generateTickets(){
 evalAll();
 if(evaluated.length<3)return [];
 const rawBudget=String(document.getElementById("budget").value||"").trim();
 const hasBudget=rawBudget!=="" && Number(rawBudget)>0;
 const budget=hasBudget?Math.max(100,Math.floor(Number(rawBudget)/100)*100):null;
 const top=evaluated[0],second=evaluated[1],third=evaluated[2],sixth=evaluated[Math.min(5,evaluated.length-1)];
 const gap12=(top?.score||0)-(second?.score||0),gap13=(top?.score||0)-(third?.score||0),gap36=(third?.score||0)-(sixth?.score||0);
 const p1=top?.place||0,p3=third?.place||0;
 let confidence=0;
 confidence+=cl(gap12/8,0,1)*0.25; confidence+=cl(gap13/12,0,1)*0.20; confidence+=cl((p1-45)/35,0,1)*0.25; confidence+=cl((p3-30)/35,0,1)*0.15; confidence+=cl(gap36/10,0,1)*0.15;
 const wideCount=Object.keys(oddsCache?.wide||{}).length;
 const pool=evaluated.slice(0,Math.min(8,evaluated.length));
 const trioAll=combos(pool,3).map(c=>{const o=trioOddsFor(c);const strength=c.reduce((s,h)=>s+h.score+h.place*.30,0);const model=c.reduce((s,h)=>s+Math.max(.04,h.place/100),0)/3;const value=o?cl(o*model/13,.55,1.7):1;return {type:"3連複",c,odds:o,utility:strength*(o?(1+cl(value-1,-.2,.35)*.22):1),value}}).sort((a,b)=>b.utility-a.utility);
 const wideAll=combos(evaluated.slice(0,Math.min(6,evaluated.length)),2).map(c=>{const o=wideOddsFor(c);const model=(c[0].place+c[1].place)/200;const strength=c.reduce((s,h)=>s+h.score+h.place*.20,0);const value=o?cl(o*model/3.0,.55,1.8):1;return {type:"ワイド",c,odds:o,utility:strength*(o?(1+cl(value-1,-.2,.4)*.25):1),value}}).sort((a,b)=>b.utility-a.utility);

 let style="3連複フォーメーション", candidates=trioAll, targetPoints=18;
 if(!hasBudget){
   if(confidence>=.68 && p1>=62){style="3連複1頭軸";targetPoints=confidence>=.78?6:10;}
   else if(confidence<.38 && wideCount>0){style="ワイド";candidates=wideAll.filter(x=>x.c.some(h=>+h.no===+top.no));targetPoints=4;}
   else {style="3連複フォーメーション";targetPoints=confidence>=.55?10:confidence>=.4?14:18;}
 }else{
   targetPoints=confidence>=.72?10:confidence>=.56?14:confidence>=.40?18:24;
   targetPoints=Math.min(targetPoints,Math.max(1,Math.floor(budget/100)));
 }

 // フォーメーションの2列目は「頭数合わせ」をしない。
 // 安定条件を満たす馬が2頭以上いる時だけフォーメーションを採用し、1頭しかいなければ1頭軸へ自動切替する。
 let formationSecond=[];
 if(style==="3連複フォーメーション"){
   const topBase=top.baseScore||top.score||0;
   const stabilityScore=h=>{
     const pop=winPopularityFor(h);
     const marketStable=pop?cl(100-(pop-1)*9,25,100):55;
     return (h.place||0)*0.55+(h.baseScore||h.score||0)*0.30+marketStable*0.15;
   };
   formationSecond=evaluated.filter(h=>+h.no!==+top.no).filter(h=>{
     const pop=winPopularityFor(h);
     const place=h.place||0,base=h.baseScore||h.score||0;
     if(place<45 || base<topBase-8)return false;
     if(pop && pop>=7)return place>=55 && base>=topBase-5;
     return true;
   }).sort((a,b)=>stabilityScore(b)-stabilityScore(a)).slice(0,2);

   if(formationSecond.length<2){
     style="3連複1頭軸";
     candidates=trioAll.filter(x=>x.c.some(h=>+h.no===+top.no));
     if(!hasBudget)targetPoints=confidence>=.72?6:confidence>=.52?10:12;
   }else{
     const secondNos=new Set(formationSecond.map(h=>+h.no));
     candidates=trioAll.filter(x=>x.c.some(h=>+h.no===+top.no) && x.c.some(h=>secondNos.has(+h.no)));
   }
 }

 if(style==="3連複1頭軸"){
   const axis=+top.no;
   // 相手頭数・点数を先に決めない。各馬を「来る余地があるか」で1頭ずつ判定し、
   // 拮抗・妙味穴・展開余地があれば紐を増やし、堅いレースなら自然に絞る。
   const rankedRivals=evaluated.filter(h=>+h.no!==axis).slice(0,7);
   const maxPts=hasBudget?Math.max(1,Math.floor(budget/100)):999;
   const tightness=cl((1-cl(gap13/11,0,1))*.45+(1-cl(gap36/11,0,1))*.35+cl((62-p1)/28,0,1)*.20,0,1);
   const scored=rankedRivals.map((h,idx)=>{
     const gap=Math.max(0,(top.score||0)-(h.score||0));
     const place=h.place||0;
     const pop=winPopularityFor(h);
     const vi=h.valueIndex||1;
     const ability=cl(1-gap/20,0,1);
     const placeScore=cl((place-22)/48,0,1);
     const valueBonus=cl((vi-1)/.45,0,1);
     const marketHole=(pop&&pop>=6)?valueBonus:0;
     const includeScore=ability*.46+placeScore*.34+marketHole*.12+tightness*.08;
     const core=(idx<=2 && gap<=14 && place>=30) || (gap<=10 && place>=38);
     const closePack=tightness>=.52 && gap<=16 && place>=28;
     const valueShot=vi>=1.15 && gap<=18 && place>=27;
     const strongValueShot=vi>=1.28 && gap<=20 && place>=24;
     return {h,idx,gap,place,pop,vi,includeScore,core,closePack,valueShot,strongValueShot,keep:core||closePack||valueShot||strongValueShot};
   });
   let selectedMetrics=scored.filter(x=>x.keep);
   // 最低限の相手は能力上位から3頭。逆に、基準を満たさない馬を点数合わせで追加しない。
   if(selectedMetrics.length<3){
     const ids=new Set(selectedMetrics.map(x=>+x.h.no));
     for(const x of scored){if(!ids.has(+x.h.no)){selectedMetrics.push(x);ids.add(+x.h.no)} if(selectedMetrics.length>=3)break;}
   }
   // 予算指定時だけ、100円×全組み合わせに収まるまで採用根拠の弱い馬から落とす。
   selectedMetrics.sort((a,b)=>b.includeScore-a.includeScore);
   while(selectedMetrics.length>3 && selectedMetrics.length*(selectedMetrics.length-1)/2>maxPts) selectedMetrics.pop();
   // 表示順はAIランキング順へ戻す。
   selectedMetrics.sort((a,b)=>a.idx-b.idx);
   const selected=selectedMetrics.map(x=>x.h);
   const rivalNos=new Set(selected.map(h=>+h.no));
   candidates=trioAll.filter(x=>{
     const nums=x.c.map(h=>+h.no);
     return nums.includes(axis) && nums.filter(v=>v!==axis).every(v=>rivalNos.has(v));
   });
   targetPoints=candidates.length;
   const oddsRows=candidates.filter(x=>x.odds&&x.odds>0);
   const valueCount=selectedMetrics.filter(x=>x.valueShot||x.strongValueShot).length;
   const closeCount=selectedMetrics.filter(x=>x.closePack).length;
   window.__lastAxisPlan={rivalCount:selected.length,points:candidates.length,tightness,valueCount,closeCount,oddsCoverage:candidates.length?oddsRows.length/candidates.length:0,selectedMetrics};
 }

 targetPoints=Math.min(targetPoints,candidates.length);
 let picks=candidates.slice(0,targetPoints);
 let suggestedBudget=budget;
 if(!hasBudget){
   const base=style==="ワイド"?300:200;
   suggestedBudget=Math.max(500,picks.length*base);
   suggestedBudget=Math.ceil(suggestedBudget/100)*100;
 }
 const stakeMap=new Array(picks.length).fill(100);
 let remaining=suggestedBudget-picks.length*100;
 const order=picks.map((x,i)=>({i,x,priority:x.utility*(x.odds?cl(x.value,.7,1.5):1)})).sort((a,b)=>b.priority-a.priority);
 for(const {i,x} of order){
   if(remaining<100)break;
   if(x.odds&&x.odds>1){
     const recover=Math.ceil((suggestedBudget*1.05/x.odds)/100)*100;
     const cap=Math.max(100,Math.floor(suggestedBudget*.4/100)*100);
     const desired=cl(recover,100,cap);
     const add=Math.min(remaining,Math.max(0,desired-stakeMap[i]));const r=Math.floor(add/100)*100;stakeMap[i]+=r;remaining-=r;
   }
 }
 let guard=0;while(remaining>=100&&picks.length&&guard<1000){const r=picks.map((x,i)=>({i,w:x.utility*(x.odds?cl(x.value,.72,1.45):1)/(1+stakeMap[i]/400)})).sort((a,b)=>b.w-a.w)[0];stakeMap[r.i]+=100;remaining-=100;guard++;}
 const total=stakeMap.reduce((a,b)=>a+b,0);
 const rows=picks.map((x,j)=>{
   const key=x.c.map(h=>h.no).sort((a,b)=>a-b).join('-');
   const ret=x.odds?Math.round((x.odds*stakeMap[j])/10)*10:null;
   const net=ret==null?null:ret-total;
   let pay='';
   if(ret!=null){pay=`<span class="small" style="margin-left:8px">${x.odds.toFixed(1)}倍 / 払戻目安 ${ret.toLocaleString()}円 / <b style="color:${net<0?'var(--d)':'var(--a)'}">${net<0?'−':'＋'}${Math.abs(net).toLocaleString()}円</b></span>`}
   else pay=`<span class="small" style="margin-left:8px;color:var(--w)">オッズ未取得</span>`;
   return `<div style="padding:6px 0;border-bottom:1px solid #2b4168"><div style="display:flex;justify-content:space-between;gap:10px;align-items:flex-start"><span>${key}${pay}</span><b>${stakeMap[j].toLocaleString()}円</b></div></div>`;
 }).join('');
 const oddsN=picks.filter(x=>x.odds).length;
 const budgetText=hasBudget?`指定予算 ${budget.toLocaleString()}円`:`AI推奨総額 ${total.toLocaleString()}円`;
 const usedNos=[...new Set(picks.flatMap(x=>x.c.map(h=>+h.no)))];
 const byRank=usedNos.slice().sort((a,b)=>evaluated.findIndex(h=>+h.no===a)-evaluated.findIndex(h=>+h.no===b));
 let structure='';
 let reason='';
 if(style==="3連複1頭軸"){
   const axis=+top.no;
   const rivals=byRank.filter(n=>n!==axis);
   structure=`<div class="card" style="margin:10px 0;background:#0c172a"><div><b>軸</b>：${axis} ${top.name}</div><div style="margin-top:6px"><b>相手</b>：${rivals.join('・')}</div></div>`;
   const ap=window.__lastAxisPlan||{};
   const raceType=(Number(ap.tightness||0)>=0.55)?"混戦":"堅め";
   reason=`<b>AI判断：${raceType}</b><br>相手${rivals.length}頭・${targetPoints}点 / 妙味採用 ${ap.valueCount||0}頭 / 拮抗採用 ${ap.closeCount||0}頭<br>${budgetText} / オッズ取得 ${oddsN}/${picks.length}点`;
 }else if(style==="3連複フォーメーション"){
   const col1=[+top.no];
   const col2=formationSecond.map(h=>+h.no);
   const col3=byRank;
   const col2Detail=formationSecond.map(h=>{const pop=winPopularityFor(h);return `${h.no}${pop?`（${pop}番人気・3着内${(h.place||0).toFixed(1)}%）`:`（3着内${(h.place||0).toFixed(1)}%）`}`}).join('・');
   structure=`<div class="card" style="margin:10px 0;background:#0c172a"><div><b>1列目</b>：${col1.join('・')}</div><div style="margin-top:6px"><b>2列目（安定重視）</b>：${col2.join('・')}</div><div class="small" style="margin-top:4px">${col2Detail}</div><div style="margin-top:6px"><b>3列目（妙味含む）</b>：${col3.join('・')}</div></div>`;
   const raceType=(gap12<=3.5&&p3>=35)?"混戦":"堅め";
   reason=`<b>AI判断：${raceType}</b><br>2列目 ${col2.length}頭 / 3列目 ${col3.length}頭 / 上位差 ${gap12.toFixed(1)}<br>${budgetText} / オッズ取得 ${oddsN}/${picks.length}点`;
 }else{
   structure=`<div class="card" style="margin:10px 0;background:#0c172a"><div><b>ワイド候補</b>：${byRank.join('・')}</div></div>`;
   reason=`<b>AI判断：混戦</b><br>ワイド ${byRank.length}頭候補 / 回収と的中のバランス重視<br>${budgetText} / オッズ取得 ${oddsN}/${picks.length}点`;
 }
 lastBetPlan={type:style,picks:picks.map((x,i)=>({numbers:x.c.map(h=>+h.no).sort((a,b)=>a-b),stake:stakeMap[i],odds:x.odds||null})),total};
 document.getElementById("ticket").innerHTML=`<div class="card" style="margin-top:10px"><b class="good">${style}・AI自動選定</b>${structure}<div class="small" style="margin:8px 0 10px">${reason}</div><div class="small" style="margin:8px 0"><b>購入内訳</b></div>${rows}<div style="margin-top:10px"><b>${picks.length}点 / 合計 ${total.toLocaleString()}円</b></div></div>`;
 return currentTickets();
}

function legacyBonus(h){
 const m=h.legacyMemory;if(!m)return 0;
 const rows=Array.isArray(m.learnedResults)?m.learnedResults:[];
 const venue=document.getElementById("venue").value,going=document.getElementById("going").value,dist=+document.getElementById("distance").value,surface=document.getElementById("surface").value;
 const bucket=d=>d<=1400?"短距離":d<=1800?"マイル中距離":d<=2200?"中距離":"長距離";
 function stat(a){const p=a.map(x=>+x.pos||0).filter(x=>x>0);if(!p.length)return 0;const avg=p.reduce((s,x)=>s+x,0)/p.length,top=p.filter(x=>x<=3).length/p.length;let b=avg<=3?2.5:avg>=8?-1.5:0;if(top>=.5)b+=1.5;return b*(p.length/(p.length+4))}
 if(rows.length){
  const c=stat(rows.filter(x=>String(x.course||"").replace(/競馬場/g,"")===venue&&String(x.surface||"")===surface));
  const g=stat(rows.filter(x=>String(x.going||"")===going));
  const d=stat(rows.filter(x=>bucket(+x.distance||0)===bucket(dist)));
  return cl(c*.40+g*.25+d*.35,-3,3);
 }
 const p=m.profile||{}, key=venue+"|"+surface;
 function fromAgg(x){if(!x||!x.n)return 0;const avg=x.sum/x.n,top=x.top3/x.n;let b=avg<=3?2.5:avg>=8?-1.5:0;if(top>=.5)b+=1.5;return b*(x.n/(x.n+4))}
 return cl(fromAgg(p.course?.[key])*.40+fromAgg(p.going?.[going])*.25+fromAgg(p.distance?.[bucket(dist)])*.35,-3,3);
}

async function jraImport(url){
   const r=await fetch(JRA_API,{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({url})});
   const j=await r.json().catch(()=>({error:"JRA取込APIの応答を読み込めません"}));
   if(!r.ok){
     const detail=j.debug?`（候補${j.debug.horse_candidates??0}頭 / links ${j.debug.anchor_count??0}）`:"";
     throw new Error((j.error||"JRA出馬表取込エラー")+detail);
  }
   try{

function historyIssues(h){
 const out=[];
 activeHistory(h).forEach((r,i)=>{
   if(r.status)return;
   const miss=[];
   if(!r.rank)miss.push("着順");
   if(!r.venue)miss.push("競馬場");
   if(!r.surface)miss.push("芝/ダ");
   if(!Number.isFinite(+r.distance)||+r.distance<=0)miss.push("距離");
   if(!r.going)miss.push("馬場");
   if(!Number.isFinite(+r.last3f))miss.push("上がり");
   if(!r.jockey)miss.push("騎手");
   if(miss.length)out.push({index:i,date:r.date||`${i+1}走前`,missing:miss});
 });
 return out;
}
function dataQuality(h){
 const hist=activeHistory(h).filter(r=>!r.status);
 if(!hist.length)return {score:0,label:"未取得",issues:[],source:"未取得"};
 let have=0,total=0;
 hist.forEach(r=>{
   const checks=[
    !!r.rank, !!r.venue, !!r.surface,
    Number.isFinite(+r.distance)&&+r.distance>0,
    !!r.going, Number.isFinite(+r.last3f), !!r.jockey
  ];
   total+=checks.length; have+=checks.filter(Boolean).length;
 });
 const score=Math.round(have/Math.max(1,total)*100);
 const issues=historyIssues(h);
 return {score,label:score===100?"完全":score>=85?"一部欠損":"要注意",issues,source:(h.history||[]).length?"netkeiba5走":"JRA4走"};
}

 if(!horses.length){if(!silent)status("histStatus","先にJRA出馬表を取り込んでください。",true);return {ok:0,total:0}}
 const already=horses.filter(h=>(h.history||[]).length).length;

   document.getElementById("histCount").textContent=`netkeiba ${already}/${horses.length}頭 / JRA照合 ${horses.filter(h=>(h.jra_history||[]).length).length}頭`;
   return {ok:already,total:horses.length};
 }
 try{
  if(!silent)status("histStatus","netkeiba5走を取得中…");
  const j=await nkFallback(horses.map(h=>h.name),document.getElementById("raceUrl").value);
  const map=new Map((j.rows||[]).map(x=>[String(x.name||x.horse_name||"").trim(),x]));let ok=0,rejected=0;
  horses.forEach(h=>{
    const r=map.get(String(h.name).trim());const rows=r?.history||r?.rows||[];
    h.netkeibaRejected=false;
    if(rows.length&&validNetkeibaHistoryForHorse(h,rows)){
      h.history=rows.slice(0,5);h.histScores=scoreLocalHistory(h.history);h.histScores.available=true;mergeNetkeibaWithJra(h);ok++;
    }else{
      h.history=[];h.histScores=null;if(rows.length){h.netkeibaRejected=true;rejected++}
    }
  });
  document.getElementById("histCount").textContent=`netkeiba ${ok}/${horses.length}頭 / JRA照合 ${horses.filter(h=>(h.jra_history||[]).length).length}頭${rejected?` / 同名別馬疑い ${rejected}頭除外`:""}`;
  renderHorses();evalAll();if(!silent)status("histStatus",`netkeiba5走 ${ok}/${horses.length}頭を取得。${rejected?`同名別馬疑い ${rejected}頭はJRA前4走へ切替。`:""}JRA4走は照合・欠損補完だけに使用します。`);return {ok,total:horses.length};
 }catch(e){if(!silent)status("histStatus",e.message||String(e),true);return {ok:0,total:horses.length}}
}
