const VENUE_CODE={"札幌":"01","函館":"02","福島":"03","新潟":"04","東京":"05","中山":"06","中京":"07","京都":"08","阪神":"09","小倉":"10"};
function raceIdFromUrl(url){
 let s=String(url||"");
 for(let i=0;i<3;i++){try{s=decodeURIComponent(s)}catch(_){break}}
 // netkeiba等の12桁race_id
 let m=s.match(/(?:race_id[=:_-]*|\/race\/)(20\d{10})/i)||s.match(/\b(20\d{10})\b/);
 if(m)return m[1];
 // JRA出馬表 accessD: ...sw01ddd10 + 競馬場2 + 年4 + 開催回2 + 開催日2 + R2 + 施行日8...
 // 例: sw01ddd1007202603040720260830 -> 202607030407
 m=s.match(/sw01ddd(?:10|01)?(\d{2})(20\d{2})(\d{2})(\d{2})(\d{2})(?:20\d{6})?/i);
 if(m)return `${m[2]}${m[1]}${m[3]}${m[4]}${m[5]}`;
 // JRA結果系。競馬場コードがURL内にあれば直接使う。
 m=s.match(/pw01(?:dde|sde)(?:10|01)?(\d{2})(20\d{2})(\d{2})(\d{2})(\d{2})(?:20\d{6})?/i);
 if(m)return `${m[2]}${m[1]}${m[3]}${m[4]}${m[5]}`;
 // 旧形式は画面の競馬場を補助的に使う。
 m=s.match(/pw01(?:dde|sde)0104(20\d{2})(\d{2})(\d{2})(\d{2})(?:20\d{6})?/i);
 if(m){
   const venue=String(raceMeta?.venue||document.getElementById("venue")?.value||"").replace(/競馬場$/,"");
   const vc=VENUE_CODE[venue]||"";
   if(vc)return `${m[1]}${vc}${m[2]}${m[3]}${m[4]}`;
 }
 return String(raceMeta?.race_id||"");
}

function oddsNumber(x){
 if(x==null)return null;
 if(Number.isFinite(+x)&&+x>0)return +x;
 if(typeof x!=="object")return null;
 const keys=["odds","win_odds","odds_min","min","value","price"];
 for(const k of keys){if(Number.isFinite(+x[k])&&+x[k]>0)return +x[k]}
 // レンジ形式なら中央値を使う（単勝の最終オッズ取得前など）。
 const lo=+[x.odds_min,x.min,x.low].find(v=>Number.isFinite(+v)&&+v>0);
 const hi=+[x.odds_max,x.max,x.high].find(v=>Number.isFinite(+v)&&+v>0);
 if(Number.isFinite(lo)&&Number.isFinite(hi))return (lo+hi)/2;
 return null;
}
