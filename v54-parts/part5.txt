ssage","field_size"];
   fields.forEach(k=>{
     const missing=(k==="distance"||k==="last3f"||k==="rank")
       ? !Number.isFinite(+r[k]) || +r[k]<=0
       : (k==="passage" ? !Array.isArray(r[k]) || !r[k].length : !r[k]);
     const hasJra=(k==="distance"||k==="last3f"||k==="rank")
       ? Number.isFinite(+jr[k]) && +jr[k]>0
       : (k==="passage" ? Array.isArray(jr[k]) && jr[k].length : !!jr[k]);
     if(missing&&hasJra){r[k]=jr[k];if(k!=="field_size")r.filled_from_jra.push(k)}
   });
 });
 h.history=nk;
 h.histScores=nk.length?scoreLocalHistory(nk):h.histScores;
 h.jraFillCount=nk.reduce((n,r)=>n+(r.filled_from_jra||[]).length,0);
 return h;
}

function scoreLocalHistory(hist){
 const targetVenue=document.getElementById("venue").value;
 const targetDistance=+document.getElementById("distance").value;
 const targetGoing=document.getElementById("going").value;
 if(!hist||!hist.length)return null;
 const w=[1,.82,.68,.56,.46];
 let formN=0,formD=0,distN=0,distD=0,courseN=0,courseD=0,goingN=0,goingD=0;
 let lastN=0,lastD=0;
 hist.slice(0,5).forEach((r,i)=>{
   if(r.status)return;
   const ww=w[i]||.5;
   const hasRank=Number.isFinite(+r.rank)&&+r.rank>0;
   const rr=hasRank?Math.max(40,100-(+r.rank-1)*5):null;
   if(rr!=null){formN+=rr*ww;formD+=ww}
   if(rr!=null && Number.isFinite(+r.distance) && +r.distance>0 && Math.abs(+r.distance-targetDistance)<=200){distN+=rr*ww;distD+=ww}
   if(rr!=null && r.venue && r.venue===targetVenue){courseN+=rr*ww;courseD+=ww}
   if(rr!=null && r.going && r.going===targetGoing){goingN+=rr*ww;goingD+=ww}
   if(Number.isFinite(+r.last3f)){lastN+=+r.last3f*ww;lastD+=ww}
 });
 const form=formD?formN/formD:65;
 const avg=lastD?lastN/lastD:null;
 const closing=avg==null?65:cl(105-(avg-33)*8,45,95);
 return {
   available:true,
   form:+form.toFixed(1),
   closing:+closing.toFixed(1),
   distance:distD?+(distN/distD).toFixed(1):70,
   course:courseD?+(courseN/courseD).toFixed(1):70,
   going:goingD?+(goingN/goingD).toFixed(1):70,
   samples:{form:+formD.toFixed(2),closing:+lastD.toFixed(2),distance:+distD.toFixed(2),course:+courseD.toFixed(2),going:+goingD.toFixed(2)},
   source:"JRA"
 };
}
