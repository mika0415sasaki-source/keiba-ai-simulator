function historyIssues(h){
 const out=[];
 activeHistory(h).forEach((r,i)=>{
   if(r.status)return;
   const miss=[];
   if(!r.rank)miss.push("着順");
   if(!r.venue)miss.push("競馬場");
   if(!r.surface)miss.push("芝/ダ");
   if(!Number.isFinite(+r.distance)||+r.distance<=0)miss.push("距離");
   if(!r.going)miss.push("馬場");
   if(!Number.isFinite(+r.last3f))miss.push("上がり");
   if(!r.jockey)miss.push("騎手");
   if(miss.length)out.push({index:i,date:r.date||`${i+1}走前`,missing:miss});
 });
 return out;
}
