async function loadNetkeibaHistories({silent=false,force=false}={}){
 if(!horses.length){if(!silent)status("histStatus","先にJRA出馬表を取り込んでください。",true);return {ok:0,total:0}}
 const already=horses.filter(h=>(h.history||[]).length).length;
 if(!force && already===horses.length){
   document.getElementById("histCount").textContent=`netkeiba ${already}/${horses.length}頭 / JRA照合 ${horses.filter(h=>(h.jra_history||[]).length).length}頭`;
   return {ok:already,total:horses.length};
 }
 try{
  if(!silent)status("histStatus","netkeiba馬DBから過去5走を再取得中…");
  const targets=force?horses:horses.filter(h=>!(h.history||[]).length);
  const j=await api("import_histories",{names:targets.map(h=>h.name),race_url:document.getElementById("raceUrl").value,target:{venue:document.getElementById("venue").value,distance:+document.getElementById("distance").value,going:document.getElementById("going").value}});
  for(const r of j.results||[]){const h=horses.find(x=>x.name===r.name);if(h){h.history=r.history||[];h.histScores=r.scores||null;h.netkeibaUrl=r.url||null;h.netkeibaVia=r.via||"";h.netkeibaError=r.error||"";mergeNetkeibaWithJra(h)}}
  const missing=horses.filter(h=>!(h.history||[]).length).map(h=>h.name);
  if(missing.length){
   try{
    const fb=await nkFallback(missing,document.getElementById("raceUrl").value);
    for(const r of fb.results||[]){const h=horses.find(x=>x.name===r.name);if(h&&r.available){h.history=r.history||[];h.histScores=r.scores||h.histScores||null;h.netkeibaUrl=r.url||null;h.netkeibaVia=r.via||"fallback";mergeNetkeibaWithJra(h)}}
   }catch(_){}
  }
  // 同名別馬の誤取得を防止。netkeiba5走がJRA前4走と1走も一致しない場合は捨て、JRAを主データに戻す。
  for(const h of horses){
    if((h.history||[]).length && !validNetkeibaHistoryForHorse(h,h.history)){
      h.netkeibaRejected=true; h.netkeibaError="JRA前4走と一致しないため同名別馬候補として除外";
      h.history=[]; h.histScores=null; h.jraFillCount=0;
    }
  }
  // fallbackがhistoryだけ返した場合もスコアを必ず生成。
  for(const h of horses){if((h.history||[]).length && !(h.histScores?.available)){h.histScores=scoreLocalHistory(h.history)}}
  renderHorses();
  const ok=horses.filter(h=>(h.history||[]).length).length;
  const jf=horses.filter(h=>!(h.history||[]).length&&(h.jra_history||[]).length).length;
  const fills=horses.reduce((n,h)=>n+(h.jraFillCount||0),0);
  document.getElementById("histCount").textContent=`netkeiba ${ok}/${horses.length}頭 / JRA照合 ${horses.filter(h=>(h.jra_history||[]).length).length}頭`;
  if(!silent)status("histStatus",`netkeiba ${ok}頭取得。JRA照合で${fills}項目を補完。${jf?`${jf}頭はJRA4走を主データとして使用します。`:""}`);
  refreshMemoryStats();
  return {ok,total:horses.length};
 }catch(e){
  const jf=horses.filter(h=>(h.jra_history||[]).length).length;
  if(!silent)status("histStatus",`netkeiba取得失敗。JRA前4走で${jf}頭を補完します。`,jf===0);
  renderHorses();
  return {ok:horses.filter(h=>(h.history||[]).length).length,total:horses.length,error:e};
 }
}
