function recordResultAudit(){
  const k=raceKey();if(!k)return false;
  const a=auditLoad(),r=a[k]||{};
  const top3=actual.slice(0,3).map(x=>+x.no).filter(Number.isFinite);
  const top3Set=new Set(top3);
  const betType=String(r.bet_type||lastBetPlan?.type||"").includes("ワイド")?"ワイド":"3連複";
  const savedTickets=Array.isArray(r.tickets)?r.tickets:[];
  const liveTickets=(lastBetPlan?.picks||[]).map(p=>(p.numbers||[]).map(Number).filter(Number.isFinite));
  const tickets=(savedTickets.length?savedTickets:liveTickets).map(t=>(t||[]).map(Number).filter(Number.isFinite));
  let hit=false;
  if(betType==="ワイド"){
    hit=tickets.some(t=>t.length===2 && t.every(n=>top3Set.has(+n)));
  }else{
    const top3Key=[...top3].sort((x,y)=>x-y).join("-");
    hit=tickets.some(t=>t.length===3 && normalizedTicketKey(t)===top3Key);
  }
  a[k]={...r,bet_type:betType,tickets,actual,verified:true,hit,result_at:new Date().toISOString()};
  if(betType==="3連複")a[k].trifecta_hit=hit;
  auditSave(a);
  markCareerResult(k,hit);
  mergedCareerStats();
  return hit
}

 const rows=Object.values(m).filter(x=>x&&x.analyzed);

 // 旧版からの累積値は「減らさない」ための下限としてだけ使う。raceMapが増えたらそちらを優先。
